Comparing my p5.js artwork with the output ChatGPT created really highlights the differences between human and AI creativity. My code was simpler and more straightforward, sticking to basic shapes and easy-to-follow logic, definitely beginner territory. ChatGPT, on the other hand, generated something more complex and polished, using advanced techniques that I hadn't even thought of yet. It felt like the AI was showing off everything it knew, while I was just trying to make something that actually worked.
When it comes to representation, it’s interesting how a coded visual portrait and a text-generated description feel so different. My sketch ended up pretty abstract and playful, more about experimenting with color and form than capturing any real likeness. The AI’s approach, though, was almost too precise, its idea was more detailed, maybe even more “realistic” in a sense, but less personal. I think that’s the main divide: my version reflects my learning process and creative quirks, while ChatGPT’s reflects a wide database of polished examples. Both are valid, but mine feels like a product of curiosity and experimentation, and the AI’s feels like a demonstration of skill and pattern recognition.


function setup() {
  createCanvas(700, 700);
  noLoop();
}

function draw() {
  background(235, 240, 245);

  drawShadow();
  drawPlate();
  drawMainFood();
  drawSides();
  drawGarnish();
}

/* ---------- DEPTH ---------- */
function drawShadow() {
  noStroke();
  fill(180, 190, 200, 100);
  ellipse(360, 420, 460, 460);
}

/* ---------- PLATE ---------- */
function drawPlate() {
  noStroke();
  fill(250);
  ellipse(350, 380, 460, 460);

  fill(230);
  ellipse(350, 380, 400, 400);
}

/* ---------- MAIN FOOD (Entree) ---------- */
function drawMainFood() {
  // Base (rice or mashed potatoes)
  fill(245, 220, 170);
  ellipse(330, 380, 260, 200);

  // Protein (grilled chicken)
  fill(210, 160, 100);
  ellipse(350, 360, 180, 110);

  // Grill lines
  stroke(170, 120, 80);
  strokeWeight(3);
  for (let i = -60; i <= 60; i += 20) {
    line(350 + i, 320, 350 + i - 15, 400);
  }
  noStroke();
}

/* ---------- SIDE DISHES ---------- */
function drawSides() {
  // Vegetables
  fill(90, 160, 100);
  ellipse(260, 420, 90, 90);
  ellipse(290, 450, 70, 70);

  // Carrots
  fill(235, 140, 60);
  rect(420, 390, 60, 18, 5);
  rect(430, 415, 50, 16, 5);
}

/* ---------- GARNISH ---------- */
function drawGarnish() {
  // Herbs
  fill(60, 130, 90);
  ellipse(340, 310, 40, 20);
  ellipse(370, 330, 35, 18);

  // Sauce drizzle
  stroke(180, 70, 70);
  strokeWeight(4);
  noFill();
  arc(350, 370, 220, 140, PI * 0.2, PI * 0.8);
  arc(350, 385, 200, 120, PI * 0.2, PI * 0.8);
}
